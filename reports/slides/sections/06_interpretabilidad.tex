% ── Slide 1: Por qué SHAP ──
\begin{frame}
\frametitle{¿Por qué SHAP?}
\textbf{Problema}: el RF Tuneado es una caja negra. Sin interpretabilidad, no es útil para política pública.

\vspace{0.3cm}
\textbf{SHAP (Lundberg \& Lee, 2017)}: descompone cada predicción como suma de contribuciones por variable:
\[
f(x_i) = \phi_0 + \sum_{j=1}^{p} \phi_{ij}
\]

La importancia global: $\text{Imp}_j = \mathbb{E}[\,|\phi_{ij}|\,]$

\vspace{0.2cm}
\textbf{Ventaja clave}: \texttt{TreeExplainer} calcula valores SHAP \textit{exactos} para árboles de decisión, sin aproximación por permutación. Válido tanto para Random Forest como para GBM.

\vspace{0.2cm}
\textit{Uso aquí}: comparar la estructura explicativa del modelo de 4 variables vs.\ el modelo de 16 variables (RF Tuneado).
\end{frame}

% ── Slide 6: Robustez ──
\begin{frame}
\frametitle{Análisis de robustez — variable objetivo alternativa}
\small
\textbf{Motivación}: \texttt{consumo\_12m} difiere de \texttt{K\_03} en 898 casos (22.6\% de los consumidores declarados).

\vspace{0.2cm}
\textbf{Hallazgo}: en la especificación de 4 variables (GBM Tuneado), reentrenar con $Y_{\text{alt}} = \mathbb{1}\{K\_03 = 1\}$ produce:

\vspace{0.2cm}
\centering
\footnotesize
\begin{tabular}{lrrr}
\toprule
Métrica & Y original & Y estricta & $\Delta$ \\
\midrule
AUC-ROC (prueba)      & 0.6959 & 0.6870 & $-0.009$ \\
CV AUC-5fold (media)  & 0.6714 & 0.6951 & $+0.027$ \\
\bottomrule
\end{tabular}

\vspace{0.25cm}
\flushleft
\scriptsize
\textbf{Conclusión}: el CV AUC \textit{mejora} con la Y estricta ($+0.027$), señal de que una operacionalización más limpia del objetivo mejora la generalización. Las relaciones estructurales son estables ante la elección de Y.\\
\textbf{Próximo paso}: reconciliar las 898 discrepancias y re-estimar el RF Tuneado con $Y_{\text{alt}}$.
\end{frame}
